cmake_minimum_required(VERSION 3.16)
project(MySync VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)
# Static linking: MSVC /MT; MinGW -static (below); Linux optional full static (glibc getaddrinfo then needs NSS at runtime, so default is static libstdc++/libgcc only)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
option(MYSYNC_FULL_STATIC "Link fully static on Linux" OFF)


# ---- third party (vendored, GPLv2-compatible licences) ----
add_library(tp_monocypher STATIC third_party/monocypher/monocypher.c)
target_include_directories(tp_monocypher PUBLIC third_party/monocypher)
add_library(tp_lz4 STATIC third_party/lz4/lz4.c)
target_include_directories(tp_lz4 PUBLIC third_party/lz4)

set(B3 third_party/blake3)
set(B3_SRC ${B3}/blake3.c ${B3}/blake3_dispatch.c ${B3}/blake3_portable.c)
add_library(tp_blake3 STATIC ${B3_SRC})
target_include_directories(tp_blake3 PUBLIC ${B3})
if(CMAKE_SYSTEM_PROCESSOR MATCHES "(x86_64|AMD64|amd64)" AND NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  if(MSVC)
    # MSVC: intrinsics files only (asm variants need MASM); per-file arch flags
    target_sources(tp_blake3 PRIVATE ${B3}/blake3_sse2.c ${B3}/blake3_sse41.c ${B3}/blake3_avx2.c ${B3}/blake3_avx512.c)
    set_source_files_properties(${B3}/blake3_sse41.c PROPERTIES COMPILE_OPTIONS "/arch:AVX")
    set_source_files_properties(${B3}/blake3_avx2.c PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
    set_source_files_properties(${B3}/blake3_avx512.c PROPERTIES COMPILE_OPTIONS "/arch:AVX512")
  else()
    target_sources(tp_blake3 PRIVATE ${B3}/blake3_sse2.c ${B3}/blake3_sse41.c ${B3}/blake3_avx2.c ${B3}/blake3_avx512.c)
    set_source_files_properties(${B3}/blake3_sse2.c PROPERTIES COMPILE_OPTIONS "-msse2")
    set_source_files_properties(${B3}/blake3_sse41.c PROPERTIES COMPILE_OPTIONS "-msse4.1")
    set_source_files_properties(${B3}/blake3_avx2.c PROPERTIES COMPILE_OPTIONS "-mavx2")
    set_source_files_properties(${B3}/blake3_avx512.c PROPERTIES COMPILE_OPTIONS "-mavx512f;-mavx512vl")
  endif()
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "(aarch64|arm64|ARM64)")
  target_sources(tp_blake3 PRIVATE ${B3}/blake3_neon.c)
  target_compile_definitions(tp_blake3 PRIVATE BLAKE3_USE_NEON=1)
else()
  target_compile_definitions(tp_blake3 PRIVATE BLAKE3_NO_SSE2 BLAKE3_NO_SSE41 BLAKE3_NO_AVX2 BLAKE3_NO_AVX512)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "(x86_64|AMD64|amd64)" AND NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  # intrinsics variants are used; do not define NO_* macros
else()
  target_compile_definitions(tp_blake3 PRIVATE BLAKE3_NO_SSE2 BLAKE3_NO_SSE41 BLAKE3_NO_AVX2 BLAKE3_NO_AVX512)
endif()
# we compile the C intrinsics, never the .S/.asm files
if(CMAKE_SYSTEM_PROCESSOR MATCHES "(x86_64|AMD64|amd64)")
  target_compile_definitions(tp_blake3 PRIVATE BLAKE3_NO_NEON)
endif()
set_target_properties(tp_blake3 tp_monocypher tp_lz4 PROPERTIES POSITION_INDEPENDENT_CODE ON)

# ---- project ----
file(GLOB CORE_SRC CONFIGURE_DEPENDS src/*.cpp)
list(REMOVE_ITEM CORE_SRC ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp)
add_library(mysync_core STATIC ${CORE_SRC})
target_include_directories(mysync_core PUBLIC include)
target_link_libraries(mysync_core PUBLIC tp_blake3 tp_monocypher tp_lz4 Threads::Threads)
if(WIN32)
  target_link_libraries(mysync_core PUBLIC ws2_32 bcrypt advapi32)
  target_compile_definitions(mysync_core PUBLIC _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
endif()
if(MSVC)
  target_compile_options(mysync_core PUBLIC /W4 /utf-8 /permissive-)
else()
  target_compile_options(mysync_core PUBLIC -Wall -Wextra -Wno-unused-parameter)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9 AND NOT WIN32)
    target_link_libraries(mysync_core PUBLIC stdc++fs)
  endif()
endif()

add_executable(mysync src/main.cpp)
target_link_libraries(mysync PRIVATE mysync_core)
if(MINGW)
  target_link_options(mysync PRIVATE -static)
endif()

enable_testing()
file(GLOB TEST_SRC CONFIGURE_DEPENDS tests/*.cpp)
add_executable(mysync_tests ${TEST_SRC})
target_link_libraries(mysync_tests PRIVATE mysync_core)
target_include_directories(mysync_tests PRIVATE tests)
if(MINGW)
  target_link_options(mysync_tests PRIVATE -static)
endif()
add_test(NAME mysync_tests COMMAND mysync_tests)

if(UNIX AND NOT APPLE)
  foreach(t mysync mysync_tests)
    if(MYSYNC_FULL_STATIC)
      target_link_options(${t} PRIVATE -static)
    else()
      target_link_options(${t} PRIVATE -static-libstdc++ -static-libgcc)
    endif()
  endforeach()
endif()
